priority: 'normal',

createTicketRecord(guild.id, {

  channelId: channel.id,
  ownerId: user.id,
  categoryKey: category.key,
  categoryName: category.name,
  staffRoleId,
  claimedBy: null,
  priority: 'normal',
  status: 'open',
  ...
});
